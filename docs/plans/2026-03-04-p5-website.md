**Step 2: Create js/validate.js**

```javascript
// js/validate.js
const API_BASE = 'https://api.opengpl.org';

async function validatePolicy() {
  const btn = document.getElementById('validate-btn');
  const input = document.getElementById('policy-input').value.trim();
  const version = document.getElementById('schema-version').value;
  const resultEl = document.getElementById('result');

  if (!input) {
    showResult(false, ['Please paste a policy before validating.']);
    return;
  }

  btn.disabled = true;
  btn.textContent = 'Validating...';
  resultEl.className = '';
  resultEl.style.display = 'none';

  try {
    const response = await fetch(`${API_BASE}/validate`, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ policy: input, version }),
    });

    if (!response.ok) {
      throw new Error(`API error: ${response.status}`);
    }

    const data = await response.json();
    showResult(data.valid, data.errors);
  } catch (err) {
    showResult(false, [`Could not reach validation API: ${err.message}`]);
  } finally {
    btn.disabled = false;
    btn.textContent = 'Validate';
  }
}

function showResult(valid, errors) {
  const el = document.getElementById('result');
  if (valid) {
    el.className = 'valid';
    el.innerHTML = '✓ Policy is valid';
  } else {
    el.className = 'invalid';
    const items = errors.map(e => `<li>${escapeHtml(e)}</li>`).join('');
    el.innerHTML = `✗ Validation failed:<ul>${items}</ul>`;
  }
}

function escapeHtml(str) {
  return str.replace(/&/g, '&amp;').replace(/</g, '&lt;').replace(/>/g, '&gt;');
}
```
